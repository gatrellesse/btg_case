# state.py — o contrato do grafo
class AgentState(BaseModel):
    document      : SourceDocument         # blocos, bbox, reader_kind, degrau, as 3 leituras
    votes         : Annotated[list[ClassifierVote], operator.add]
    classification: Classification | None  # saída do consenso
    candidates    : Annotated[list[tuple[str, FieldCandidate]], operator.add]
    fields        : dict[str, ExtractedField]
    validations   : list[ValidationResult] # nome, args e retorno de cada tool
    triage        : Triage | None
    # contadores de ciclo — a única defesa contra loop infinito
    reprompt_count: int = 0
    repair_count  : int = 0
    # mesmo reducer, mesmo motivo: dois nós do fan-out podem falhar no
    # mesmo tick, e duas falhas simultâneas não podem derrubar o documento
    errors        : Annotated[list[NodeError], operator.add]
